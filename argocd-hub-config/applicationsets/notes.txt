# clusters/nane1/applicationset.yaml
#---------------------
apiVersion: argoproj.io/v1alpha1
kind: ApplicationSet
metadata:
  name: nane1-applications
  namespace: argocd
spec:
  goTemplate: true
  goTemplateOptions: ["missingkey=error"]

  generators:
    - git:
        repoURL: https://github.com/myapporg/argocd-applicationset
        revision: main
        files:
          - path: tenants/*.yaml

  template:
    metadata:
      name: '{{.appCode}}-{{.deployment.env}}'
    spec:
      project: '{{.appCode}}'
      source:
        repoURL: '{{.repoURL}}'
        targetRevision: HEAD
        path: '{{.chartPath}}'
        helm:
          valueFiles:
            - '{{.deployment.valuesFile}}'
      destination:
        server: https://kubernetes.default.svc
        namespace: '{{.deployment.namespace}}'
      syncPolicy:
        automated:
          prune: true
          selfHeal: true

  templatePatch: |
    {{- if ne .deployment.cluster "nane1" }}
    spec:
      syncPolicy: {}
    {{- end }}

---

apiVersion: argoproj.io/v1alpha1
kind: ApplicationSet
metadata:
  name: projects
spec:
  generators:
    - git:
        repoURL: https://github.com/myapporg/argocd-applicationset
        revision: main
        files:
          - path: tenants/*.yaml

  template:
    metadata:
      name: '{{appCode}}-project'
    spec:
      project: default
      source:
        repoURL: https://github.com/myapporg/argocd-applicationset

        path: project-template
      destination:
        server: https://kubernetes.default.svc
        namespace: argocd

project-template/project.yaml

apiVersion: argoproj.io/v1alpha1
kind: AppProject
metadata:
  name: '{{appCode}}'
spec:
  description: Project for {{appCode}}
  sourceRepos:
    - '{{repoURL}}'
  destinations:
    - namespace: '{{appCode}}-*'
      server: '*'
  clusterResourceWhitelist:
    - group: '*'
      kind: '*'

argocd Bootstrap applicationset
bootstrap-applicationset.yaml

Add this to your bootstrap repo:

apiVersion: argoproj.io/v1alpha1
kind: ApplicationSet
metadata:
  name: bootstrap-applicationsets
  namespace: argocd
spec:
  generators:
    - git:
        repoURL: https://github.com/myapporg/argocd-applicationset
        revision: main
        directories:
          - path: clusters/*

  template:
    metadata:
      name: '{{path.basename}}-apps'
    spec:
      project: default
      source:
        repoURL: https://github.com/myapporg/argocd-applicationset
        targetRevision: main
        path: '{{path}}'
      destination:
        server: https://kubernetes.default.svc
        namespace: argocd
      syncPolicy:
        automated:
          prune: true
          selfHeal: true

What this does
Watches:
clusters/nane1/
clusters/us-central1/
For each folder → creates an ArgoCD Application
That Application contains your real ApplicationSets.
This is your App-of-Apps → ApplicationSet bridge
